# ==========================================================================================
# CT_toroidal 剖面输入文件（示例）—— 本程序**唯一**的剖面来源

# 格式: 每行一条 "名字 = 数值";  '#' 之后到行尾为注释;  空行忽略。
# 读取: cfg.profile_file = 'profile_example.txt';  out = CT_toroidal(cfg);
#       (文件不在当前目录时, 程序会自动到本程序目录里找; 也支持绝对路径)
#
# 【必需字段】缺任何一个都会报错, 并一次性列出全部缺失项:
#   rho  R0  B0  a  q  shear  A_ion  theta  n_e  T_e
#   以及梯度四项: RL_ne RL_Te RL_ni RL_Ti  (或用 dn_e_dr/dT_e_dr/dn_i_dr/dT_i_dr 代替)
# 【可选字段】不写则取默认值:
#   kappa=1.00 ;  r=rho*a ;  n_i=n_e ;  T_i=T_e
#
# 【严格规则】(违反即报错, 报错带行号)
#   * 字段名拼错 -> 报错, 并给出最接近的字段名建议
#   * 同一字段写两次 -> 报错
#   * 值不是数 / 值为空 -> 报错
#   * 某行没有 "=" -> 报错
#   * 同一物理量同时给 RL_X 与 dX_dr -> 报错(二义)
#   * rho 需在 (0,1); R0,B0,a,q,A_ion,kappa,n,T 需 >0; theta 需在 [-pi,pi];
#     0 < r < a; RL_ne/RL_Te/RL_ni/RL_Ti 需 >0
#   * 同时给 r 与 rho 时必须满足 |r - rho*a| <= 0.05*a
# ==========================================================================================

# ------------------------------- A. 磁面与磁位形 -------------------------------
rho     = 0.45          # r/a                     [-]
R0      = 1.85          # 大半径                   [m]
B0      = 2.50          # 轴上环向磁场             [T]
a       = 0.45          # 小半径                   [m]
q       = 2          # 安全因子                 [-]
shear   = 0.92          # s_hat = (r/q) dq/dr      [-]
A_ion   = 2.0           # 主离子质量数(D=2)        [-]
theta   = 0.0           # 极向角(0 = 外中平面)     [rad]
# kappa = 1.00          # 拉长比(本版按圆截面处理; 可选)
# r     = 0.279         # 小半径 [m]; 不写则 r = rho*a; 若给则须与 rho*a 一致(5% 内)

# ------------------------------- B. 局部剖面量 ---------------------------------
n_e     = 3.2e19        # 局部电子密度             [m^-3]
T_e     = 1500          # 局部电子温度             [eV]
n_i     = 3.2e19        # 局部主离子密度(可选)     [m^-3]
T_i     = 850           # 局部主离子温度(可选)     [eV]

RL_ne   = 3.15           # R/L_ne   (rho=0.62 处实验值)

RL_Ti   = 2.0          # R/L_Ti   ==> eta_i = 2
#
# 方式二: 只给物理梯度(单位 SI), 程序自动反算 R/L_X = -R0 (dX/dr)/X。
#         注意: 同一物理量**不能**同时给 RL_X 与 dX_dr(会报错);

# dn_i_dr = -2.3e20     # dn_i/dr  [m^-4]
# dT_i_dr = -5.45e3     # dT_i/dr  [eV/m]
